function configurarBaseDeDatos() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const estructuras = [
    { nombre: "Evaluaciones", columnas: ["Timestamp", "Fecha", "Curso", "Asignatura", "Tipo", "Profesor", "Detalles"] },
    { nombre: "Config_Cursos", columnas: ["Curso", "Max_Escritas_Semana", "Max_Total_Semana", "Max_Por_Dia", "Max_Otras_Dia"] },
    // 👇 AQUÍ AJUSTAMOS LAS 4 COLUMNAS DE HORAS
    { nombre: "Config_Asignaturas", columnas: ["Asignatura", "Color_Hex", "Ignorar_Limites", "", "", "", "Hrs_1er_Ciclo", "Hrs_2do_Ciclo", "Hrs_Media_1", "Hrs_Media_2"] },
    { nombre: "Config_Bloques", columnas: ["Nombre_Bloque", "Tipo", "Inicio", "Fin"] }, 
    { nombre: "Periodos_Evaluacion", columnas: ["Nombre_Periodo", "Fecha_Inicio", "Fecha_Fin", "Color_Fondo_Hex"] },
    { nombre: "Carga_Automatica", columnas: ["Asignatura", "Fecha_Inicio", "Fecha_Fin", "Tipo", "Activo"] },
    { nombre: "horarios_cursos", columnas: ["Curso", "Día", "Rango Horario", "Rango Horario.1", "Asignatura 1", "Asignatura 2", "Profesor"] },
    { nombre: "Usuarios_Autorizados", columnas: ["Email", "Rol"] },
    { nombre: "Config_General", columnas: ["Parametro", "Valor", "", "", "", "", "Fechas_Bloqueadas"] },
    { nombre: "db_materias_por_curso", columnas: ["Curso", "Asignatura_Excluida"] } 
  ];
  
  estructuras.forEach(pestaña => {
    let sheet = ss.getSheetByName(pestaña.nombre);
    if (!sheet) {
      sheet = ss.insertSheet(pestaña.nombre);
      sheet.getRange(1, 1, 1, pestaña.columnas.length).setValues([pestaña.columnas]).setFontWeight("bold");
      sheet.setFrozenRows(1);
      if (pestaña.nombre === "Config_General") {
        sheet.appendRow(["Dias_Anticipacion", 7]);
        sheet.appendRow(["Sistema_Abierto", "SI"]); 
      }
    } else {
      sheet.getRange(1, 1, 1, pestaña.columnas.length).setValues([pestaña.columnas]).setFontWeight("bold");
    }
  });
  return "Base de datos configurada.";
}

function verificarPermisos() {
  const email = Session.getActiveUser().getEmail().toLowerCase();
  if (!email) throw new Error("No se pudo identificar tu cuenta de Google.");
  
  const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName('Usuarios_Autorizados');
  if (!sheet) throw new Error("Falta la pestaña 'Usuarios_Autorizados' en el Excel.");
  
  const data = sheet.getDataRange().getValues().slice(1);
  let rolAsignado = null;

  for (let i = 0; i < data.length; i++) {
    if (data[i][0].toString().trim().toLowerCase() === email) {
      rolAsignado = data[i][1] ? data[i][1].toString().trim() : 'Profesor';
      break;
    }
  }

  if (!rolAsignado) throw new Error(`Acceso denegado: El correo ${email} no tiene permisos de acceso.`);
  return { email: email, rol: rolAsignado }; 
}

function getEstadoUsuario() {
  const email = Session.getActiveUser().getEmail().toLowerCase();
  
  const sheetGen = SpreadsheetApp.getActiveSpreadsheet().getSheetByName('Config_General');
  let abierto = true;
  if (sheetGen) {
    const data = sheetGen.getDataRange().getValues();
    const filaAbierto = data.find(r => r[0] === 'Sistema_Abierto');
    if (filaAbierto && filaAbierto[1].toString().trim().toUpperCase() === 'NO') abierto = false;
  }

  try {
    const usuario = verificarPermisos();
    return { autorizado: true, email: usuario.email, rol: usuario.rol, sistemaAbierto: abierto };
  } catch(e) {
    return { autorizado: false, email: email, rol: 'Ninguno', sistemaAbierto: abierto };
  }
}

function getDataConfig() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const cursos = ss.getSheetByName('Config_Cursos').getDataRange().getValues().slice(1).map(r => r[0]).filter(String);
  
  const asignaturas = ss.getSheetByName('Config_Asignaturas').getDataRange().getValues().slice(1).map(r => ({
    nombre: r[0], color: r[1], hrs1: r[6], hrs2: r[7], hrsM1: r[8], hrsM2: r[9]
  })).filter(a => a.nombre);
  
  let bloques = [];
  const sheetBloques = ss.getSheetByName('Config_Bloques');
  if (sheetBloques) {
    bloques = sheetBloques.getDataRange().getDisplayValues().slice(1).map(r => ({
      nombre: r[0], tipo: r[1].toString().trim().toLowerCase() === 'recreo' ? 'recreo' : 'clase',
      inicio: r[2], fin: r[3]
    })).filter(b => b.inicio && b.fin);
  }

  let materiasExcluidas = [];
  const sheetExcluidas = ss.getSheetByName('db_materias_por_curso');
  if (sheetExcluidas) {
    materiasExcluidas = sheetExcluidas.getDataRange().getValues().slice(1).map(r => ({
      curso: r[0].toString().trim().toUpperCase(),
      asignatura: r[1].toString().trim().toUpperCase()
    })).filter(ex => ex.curso && ex.asignatura);
  }

  // Lectura de parámetros dinámicos del Artículo 42
  let art42 = { h1_min: 2, h1_max: 4, h2_min: 3, h2_max: 5, h3_min: 4, h3_max: 7 };
  const sheetGen = ss.getSheetByName('Config_General');
  if (sheetGen) {
    sheetGen.getDataRange().getValues().forEach(r => {
      if(r[0] === 'Art42_Min_1a2_Hrs') art42.h1_min = parseInt(r[1]) || 2;
      if(r[0] === 'Art42_Max_1a2_Hrs') art42.h1_max = parseInt(r[1]) || 4;
      if(r[0] === 'Art42_Min_3a4_Hrs') art42.h2_min = parseInt(r[1]) || 3;
      if(r[0] === 'Art42_Max_3a4_Hrs') art42.h2_max = parseInt(r[1]) || 5;
      if(r[0] === 'Art42_Min_5_Hrs') art42.h3_min = parseInt(r[1]) || 4;
      if(r[0] === 'Art42_Max_5_Hrs') art42.h3_max = parseInt(r[1]) || 7;
    });
  }

  return { cursos, asignaturas, bloques, materiasExcluidas, art42 };
}


function excluirMateriasBackend(lista) {
  const usuario = verificarPermisos();
  if (usuario.rol.toLowerCase() !== 'admin' && usuario.rol.toLowerCase() !== 'administrador') throw new Error("Solo administradores.");
  
  const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName('db_materias_por_curso');
  if (!sheet) throw new Error("Ejecuta configurarBaseDeDatos() primero para crear la pestaña.");
  
  const nuevasFilas = lista.map(item => [item.curso, item.asignatura]);
  sheet.getRange(sheet.getLastRow() + 1, 1, nuevasFilas.length, 2).setValues(nuevasFilas);
  return "Asignaturas excluidas correctamente.";
}

function restaurarMateriasBackend() {
  const usuario = verificarPermisos();
  if (usuario.rol.toLowerCase() !== 'admin' && usuario.rol.toLowerCase() !== 'administrador') throw new Error("Solo administradores.");
  
  const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName('db_materias_por_curso');
  if (sheet && sheet.getLastRow() > 1) sheet.deleteRows(2, sheet.getLastRow() - 1);
  return "Todas las exclusiones han sido borradas.";
}

// ACTUALIZACIÓN: Ahora soporta filtros múltiples (Arrays) para la exportación PDF
function obtenerEventosCalendario(filtroCursos, filtroAsigs) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const tz = ss.getSpreadsheetTimeZone();
  let todosLosEventos = [];

  // Adaptación de filtros (Strings a Arrays si es necesario)
  let cursosPermitidos = [];
  if (filtroCursos) cursosPermitidos = Array.isArray(filtroCursos) ? filtroCursos : [filtroCursos];
  
  let asigsPermitidas = [];
  if (filtroAsigs) asigsPermitidas = Array.isArray(filtroAsigs) ? filtroAsigs : [filtroAsigs];

  const sheetPeriodos = ss.getSheetByName('Periodos_Evaluacion');
  if (sheetPeriodos) {
    sheetPeriodos.getDataRange().getValues().slice(1).forEach(fila => {
      if (fila[1] && fila[2]) {
        const fInicio = new Date(fila[1]); const fFin = new Date(fila[2]); fFin.setDate(fFin.getDate() + 1); 
        todosLosEventos.push({
          title: fila[0], start: Utilities.formatDate(fInicio, tz, "yyyy-MM-dd"), end: Utilities.formatDate(fFin, tz, "yyyy-MM-dd"),
          display: 'background', backgroundColor: fila[3] || '#fff3cd'
        });
      }
    });
  }

  const sheetGen = ss.getSheetByName('Config_General');
  if (sheetGen) {
    sheetGen.getDataRange().getValues().slice(1).forEach(fila => {
      if (fila[6]) { 
        const fechaObj = new Date(fila[6]);
        if (!isNaN(fechaObj)) {
          todosLosEventos.push({
            title: '🚫 Día Bloqueado', start: Utilities.formatDate(fechaObj, tz, "yyyy-MM-dd"),
            allDay: true, backgroundColor: '#dc3545', borderColor: '#dc3545', extendedProps: { bloqueado: true }
          });
        }
      }
    });
  }

  const sheetEval = ss.getSheetByName('Evaluaciones');
  if (sheetEval) {
    const configAsignaturas = ss.getSheetByName('Config_Asignaturas').getDataRange().getValues().slice(1);
    const colores = {}; configAsignaturas.forEach(r => colores[r[0]] = r[1]);

    const eventosEvaluaciones = sheetEval.getDataRange().getValues().slice(1).filter(fila => {
        if(!fila[1]) return false; 
        
        // VALIDACIÓN DE FILTROS MÚLTIPLES
        const cumpleCurso = cursosPermitidos.length === 0 || cursosPermitidos.includes(fila[2].toString().trim());
        const cumpleAsig = asigsPermitidas.length === 0 || asigsPermitidas.includes(fila[3].toString().trim());
        
        return cumpleCurso && cumpleAsig;
      }).map(fila => {
        const fechaStr = Utilities.formatDate(new Date(fila[1]), tz, "yyyy-MM-dd");
        let titulo = `${fila[2]} - ${fila[3]}`;
        if (fila[6] && fila[6].toString().includes("⏱️")) titulo += ` | ${fila[6].toString().split(" | ")[0]}`;
        return {
          title: titulo, start: fechaStr, backgroundColor: colores[fila[3]] || '#3788d8',
          extendedProps: { profesor: fila[5], tipo: fila[4], detalles: fila[6], curso: fila[2], asignatura: fila[3] }
        };
      });
    todosLosEventos = todosLosEventos.concat(eventosEvaluaciones);
  }
  return todosLosEventos;
}

function agendarDesdeCalendario(datos) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const timezone = ss.getSpreadsheetTimeZone();
  const usuario = verificarPermisos(); 
  const emailProfesor = usuario.email;
  
  const configGen = ss.getSheetByName('Config_General');
  let diasAnticipacion = 0;
  if (configGen) {
    const dataGen = configGen.getDataRange().getValues();
    const filaAbierto = dataGen.find(r => r[0] === 'Sistema_Abierto');
    if (filaAbierto && filaAbierto[1].toString().trim().toUpperCase() === 'NO') {
      throw new Error("El periodo de agendamiento se encuentra cerrado temporalmente por administración.");
    }
    const filaAnt = dataGen.find(r => r[0] === 'Dias_Anticipacion');
    if (filaAnt && !isNaN(filaAnt[1])) diasAnticipacion = parseInt(filaAnt[1]);

    for (let r of dataGen.slice(1)) {
      if (r[6]) {
        const fechaBloqueada = Utilities.formatDate(new Date(r[6]), timezone, "yyyy-MM-dd");
        if (fechaBloqueada === datos.fecha) {
          throw new Error("No se puede agendar en esta fecha. El día se encuentra bloqueado por administración.");
        }
      }
    }
  }

  if (diasAnticipacion > 0) {
    const hoy = new Date(); hoy.setHours(0,0,0,0);
    const fechaCita = new Date(datos.fecha + "T00:00:00");
    const diffDias = Math.floor((fechaCita.getTime() - hoy.getTime()) / (1000 * 3600 * 24));
    if (diffDias < diasAnticipacion) throw new Error(`Debes agendar con al menos ${diasAnticipacion} días de anticipación.`);
  }

  const fechaNueva = new Date(datos.fecha + "T12:00:00"); 
  if (fechaNueva.getDay() === 0 || fechaNueva.getDay() === 6) throw new Error("No se agendan evaluaciones los fines de semana.");

  const dataCursos = ss.getSheetByName('Config_Cursos').getDataRange().getValues();
  let maxEscritasSemana = 2, maxTotalSemana = 4, maxEscritasDia = 2, maxOtrasDia = 2; 
  const cursoBuscado = datos.curso.toString().trim(); 

  // OBTENER ASIGNATURAS EXENTAS
  const asignaturasExentas = [];
  const configAsig = ss.getSheetByName('Config_Asignaturas').getDataRange().getValues().slice(1);
  configAsig.forEach(r => {
    if (r[2] && r[2].toString().trim().toUpperCase() === 'SI') asignaturasExentas.push(r[0].toString().trim());
  });
  const esExenta = asignaturasExentas.includes(datos.asignatura.toString().trim());

  for (let i = 1; i < dataCursos.length; i++) {
    if (dataCursos[i][0].toString().trim() === cursoBuscado) {
      maxEscritasSemana = parseInt(dataCursos[i][1]) || maxEscritasSemana;
      maxTotalSemana = parseInt(dataCursos[i][2]) || maxTotalSemana;
      maxEscritasDia = parseInt(dataCursos[i][3]) || maxEscritasDia;
      maxOtrasDia = parseInt(dataCursos[i][4]) || maxEscritasDia; 
      break;
    }
  }

  const evaluaciones = ss.getSheetByName('Evaluaciones').getDataRange().getValues().slice(1);
  let conteoDiaEscritas = 0, conteoDiaOtras = 0, conteoSemanaTotal = 0, conteoSemanaEscritas = 0;

  const getWeek = (d) => {
    const date = new Date(d.getTime()); date.setHours(0,0,0,0); date.setDate(date.getDate() + 4 - (date.getDay() || 7));
    return Math.ceil((((date - new Date(date.getFullYear(),0,1)) / 8.64e7) + 1)/7);
  };
  const numSemanaNueva = getWeek(fechaNueva); const añoNuevo = fechaNueva.getFullYear();

  evaluaciones.forEach(fila => {
    if (fila[2] && fila[2].toString().trim() === cursoBuscado && fila[1]) { 
      
      // IGNORAR MATERIAS EXENTAS EN EL CONTEO
      if (asignaturasExentas.includes(fila[3].toString().trim())) return; 

      const fechaFilaObj = new Date(fila[1]);
      if (getWeek(fechaFilaObj) === numSemanaNueva && fechaFilaObj.getFullYear() === añoNuevo) {
        conteoSemanaTotal++;
        if (fila[4] === 'Escrita') conteoSemanaEscritas++;
      }
      if (Utilities.formatDate(fechaFilaObj, timezone, "yyyy-MM-dd") === datos.fecha) {
        if (fila[4] === 'Escrita') conteoDiaEscritas++; else conteoDiaOtras++;
      }
    }
  });

  if (!esExenta) {
    if (datos.tipo === 'Escrita' && conteoDiaEscritas >= maxEscritasDia) throw new Error(`Límite diario de pruebas escritas superado (${maxEscritasDia}).`);
    if (datos.tipo !== 'Escrita' && conteoDiaOtras >= maxOtrasDia) throw new Error(`Límite diario de trabajos superado (${maxOtrasDia}).`);
    if (conteoSemanaTotal >= maxTotalSemana) throw new Error(`Límite semanal alcanzado para ${datos.curso}.`);
    if (datos.tipo === 'Escrita' && conteoSemanaEscritas >= maxEscritasSemana) throw new Error(`Límite de pruebas escritas semanales alcanzado.`);
  }

  ss.getSheetByName('Evaluaciones').appendRow([new Date(), datos.fecha, datos.curso, datos.asignatura, datos.tipo, emailProfesor, datos.detalles || "Ingreso Manual"]);
  return "Evaluación agendada exitosamente.";
}

function eliminarEvaluacionBackend(datos) {
  const usuario = verificarPermisos();
  const esAdmin = (usuario.rol.toLowerCase() === 'admin' || usuario.rol.toLowerCase() === 'administrador');
  
  if (!esAdmin && usuario.email.toLowerCase() !== datos.profesor.toLowerCase()) {
    throw new Error("Solo el autor original o un Administrador puede eliminar esta evaluación.");
  }

  const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName('Evaluaciones');
  const data = sheet.getDataRange().getValues();
  const tz = SpreadsheetApp.getActiveSpreadsheet().getSpreadsheetTimeZone();
  
  let eliminada = false;
  for (let i = data.length - 1; i >= 1; i--) {
    const fila = data[i];
    if (!fila[1]) continue;
    const fechaStr = Utilities.formatDate(new Date(fila[1]), tz, "yyyy-MM-dd");
    
    if (fechaStr === datos.fecha && fila[2].toString().trim() === datos.curso.trim() && fila[3].toString().trim() === datos.asignatura.trim() && fila[5].toString().trim().toLowerCase() === datos.profesor.toLowerCase()) {
        sheet.deleteRow(i + 1); 
        eliminada = true;
        break; 
    }
  }
  if (!eliminada) throw new Error("No se encontró la evaluación o ya fue eliminada.");
  return "La evaluación ha sido eliminada correctamente.";
}

function editarEvaluacionBackend(datosAntiguos, datosNuevos) {
  const usuario = verificarPermisos();
  const esAdmin = (usuario.rol.toLowerCase() === 'admin' || usuario.rol.toLowerCase() === 'administrador');
  const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName('Evaluaciones');
  const data = sheet.getDataRange().getValues();

  for (let i = 1; i < data.length; i++) {
    let r = data[i];
    let fStr = r[1] instanceof Date ? Utilities.formatDate(r[1], Session.getScriptTimeZone(), "yyyy-MM-dd") : r[1].toString().split('T')[0];
    
    if (fStr === datosAntiguos.fecha && r[2] === datosAntiguos.curso && r[3] === datosAntiguos.asignatura) {
      if (!esAdmin && r[5].toLowerCase() !== usuario.email.toLowerCase()) {
        throw new Error("No tienes permiso para editar esta evaluación.");
      }
      sheet.getRange(i + 1, 2).setValue(datosNuevos.fecha);
      sheet.getRange(i + 1, 3).setValue(datosNuevos.curso);
      sheet.getRange(i + 1, 4).setValue(datosNuevos.asignatura);
      sheet.getRange(i + 1, 5).setValue(datosNuevos.tipo);
      sheet.getRange(i + 1, 7).setValue(datosNuevos.detalles);
      return "Evaluación actualizada correctamente.";
    }
  }
  throw new Error("No se encontró la evaluación original.");
}


function reclamarAutoriaBackend(datosEvaluacion) {
  const usuario = verificarPermisos(); // Usa la función que ya tienes para obtener al usuario
  const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName('Evaluaciones');
  const data = sheet.getDataRange().getValues();

  for (let i = 1; i < data.length; i++) {
    let r = data[i];
    let fStr = r[1] instanceof Date ? Utilities.formatDate(r[1], Session.getScriptTimeZone(), "yyyy-MM-dd") : r[1].toString().split('T')[0];
    
    // Verifica que coincida la evaluación y que el dueño original sea UTP
    if (fStr === datosEvaluacion.fecha && r[2] === datosEvaluacion.curso && r[3] === datosEvaluacion.asignatura && r[5].toString().toUpperCase() === 'COORDINACION PEDAGOGICA') {
      sheet.getRange(i + 1, 6).setValue(usuario.email); // Reemplaza el profesor por el email actual (Columna F)
      return "La evaluación ha sido transferida a tu usuario exitosamente.";
    }
  }
  throw new Error("No se pudo transferir. Es posible que la evaluación ya no exista o ya tenga otro dueño.");
}

function obtenerHorarioCurso(cursoBuscado) {
  const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName('horarios_cursos');
  if (!sheet) return [];
  return sheet.getDataRange().getDisplayValues().slice(1).filter(r => r[0].toString().trim() === cursoBuscado);
}

function guardarHorarioCurso(curso, filasHorario) {
  const usuario = verificarPermisos(); 
  if (usuario.rol.toLowerCase() !== 'admin' && usuario.rol.toLowerCase() !== 'administrador') {
    throw new Error("Acceso denegado: Solo los administradores pueden modificar la estructura de horarios.");
  }

  const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName('horarios_cursos');
  const datos = sheet.getDataRange().getValues();
  for (let i = datos.length - 1; i >= 1; i--) {
    if (datos[i][0].toString().trim() === curso) sheet.deleteRow(i + 1);
  }
  
  if (filasHorario && filasHorario.length > 0) {
    const nuevosDatos = filasHorario.map(fila => [curso, fila.dia, fila.inicio, fila.fin, fila.asig1, fila.asig2, fila.profesor]);
    sheet.getRange(sheet.getLastRow() + 1, 1, nuevosDatos.length, nuevosDatos[0].length).setValues(nuevosDatos);
  }
  return `Horario de ${curso} guardado exitosamente.`;
}

function ejecutarCargaAutomaticaBackend(permitirCortas) {
  const usuario = verificarPermisos(); 
  if (usuario.rol.toLowerCase() !== 'admin' && usuario.rol.toLowerCase() !== 'administrador') {
    throw new Error("Acceso denegado: Solo los administradores pueden ejecutar cargas masivas automáticas.");
  }

  const ss = SpreadsheetApp.getActiveSpreadsheet(); const ui = SpreadsheetApp.getUi(); const tz = ss.getSpreadsheetTimeZone();
  const sheetCarga = ss.getSheetByName('Carga_Automatica'); const sheetHorarios = ss.getSheetByName('horarios_cursos');
  const sheetEval = ss.getSheetByName('Evaluaciones'); const sheetCursos = ss.getSheetByName('Config_Cursos');
  const sheetGen = ss.getSheetByName('Config_General');
  
  if (!sheetHorarios || !sheetCarga) throw new Error("Faltan pestañas necesarias.");

  const fechasBloqueadas = new Set();
  if (sheetGen) {
    sheetGen.getDataRange().getValues().slice(1).forEach(r => {
      if (r[6]) {
        const d = new Date(r[6]);
        if (!isNaN(d)) fechasBloqueadas.add(Utilities.formatDate(d, tz, "yyyy-MM-dd"));
      }
    });
  }

  // OBTENER ASIGNATURAS EXENTAS PARA LA CARGA AUTOMÁTICA
  const asignaturasExentas = [];
  const sheetAsigConf = ss.getSheetByName('Config_Asignaturas');
  if (sheetAsigConf) {
    sheetAsigConf.getDataRange().getValues().slice(1).forEach(r => {
      if (r[2] && r[2].toString().trim().toUpperCase() === 'SI') asignaturasExentas.push(r[0].toString().trim());
    });
  }

  const datosHorarios = sheetHorarios.getDataRange().getValues().slice(1);
  const mapaHorarios = {};
  
  datosHorarios.forEach(fila => {
    const curso = fila[0].toString().trim(); const dia = fila[1].toString().trim().toUpperCase();
    const inicio = fila[2]; const fin = fila[3];    
    const asig1 = fila[4] ? fila[4].toString().trim().toUpperCase() : ""; 
    const asig2 = fila[5] ? fila[5].toString().trim().toUpperCase() : ""; 
    const esCorta = (asig1 !== "" && asig2 !== ""); 
    
    if (!mapaHorarios[curso]) mapaHorarios[curso] = {};
    if (asig1) { if (!mapaHorarios[curso][asig1]) mapaHorarios[curso][asig1] = []; mapaHorarios[curso][asig1].push({ dia, inicio, fin, esCorta }); }
    if (asig2) { if (!mapaHorarios[curso][asig2]) mapaHorarios[curso][asig2] = []; mapaHorarios[curso][asig2].push({ dia, inicio, fin, esCorta }); }
  });

  const limitesCursos = {};
  sheetCursos.getDataRange().getValues().slice(1).forEach(r => { limitesCursos[r[0].toString().trim()] = { maxEscritas: r[1], maxTotal: r[2], maxDia: r[3], maxOtras: r[4] }; });
  
  const evaluacionesExistentes = sheetEval.getDataRange().getValues().slice(1);
  const reglas = sheetCarga.getDataRange().getValues().slice(1).filter(r => r[4] && r[4].toString().trim().toLowerCase() === 'si'); 
    
  let nuevasEvaluaciones = []; let logErrores = [];
  const mapaDiasNumeros = { 1: "LUNES", 2: "MARTES", 3: "MIERCOLES", 4: "JUEVES", 5: "VIERNES" };

  reglas.forEach(regla => {
    const asigRegla = regla[0].toString().trim().toUpperCase(); const fInicio = new Date(regla[1]); const fFin = new Date(regla[2]); const tipo = regla[3].toString().trim();
    if (isNaN(fInicio) || isNaN(fFin)) return;
    fInicio.setHours(0,0,0,0); fFin.setHours(23,59,59,999);

    for (const curso in mapaHorarios) {
      if (mapaHorarios[curso][asigRegla]) {
        const bloquesClase = mapaHorarios[curso][asigRegla]; 
        let agendado = false; let iterador = new Date(fInicio.getTime()); let limite = new Date(fInicio.getTime()); limite.setDate(limite.getDate() + 120); 
        
        while (!agendado && iterador <= limite) {
          const fechaStrIter = Utilities.formatDate(iterador, tz, "yyyy-MM-dd");
          
          if (fechasBloqueadas.has(fechaStrIter)) {
            iterador.setDate(iterador.getDate() + 1);
            continue;
          }

          const diaTxt = mapaDiasNumeros[iterador.getDay()];
          const bloquesHoy = bloquesClase.filter(b => b.dia === diaTxt);

          if (diaTxt && bloquesHoy.length > 0) {
            let bElegido = null;
            for (let b of bloquesHoy) { if (!permitirCortas && b.esCorta) continue; bElegido = b; break; }
            
            // PASAMOS LAS ASIGNATURAS EXENTAS AL VERIFICADOR
            if (bElegido && !verificarTopesEnMemoria(curso, iterador, tipo, limitesCursos, evaluacionesExistentes, nuevasEvaluaciones, asigRegla, asignaturasExentas)) {
                let msj = `${calcularDuracion(bElegido.inicio, bElegido.fin, tz, bElegido.esCorta)} | Auto`;
                if (iterador > fFin) msj += " (Fuera Rango)";
                nuevasEvaluaciones.push([new Date(), fechaStrIter, curso, asigRegla, tipo, "SISTEMA AUTOMÁTICO", msj]);
                agendado = true; break; 
            }
          }
          iterador.setDate(iterador.getDate() + 1);
        }
        if (!agendado) logErrores.push(`Sin cupo para ${asigRegla} en ${curso}.`);
      }
    }
  });

  if (nuevasEvaluaciones.length > 0) {
    sheetEval.getRange(sheetEval.getLastRow() + 1, 1, nuevasEvaluaciones.length, nuevasEvaluaciones[0].length).setValues(nuevasEvaluaciones);
    ui.alert(`✅ Carga finalizada.\nAgendadas: ${nuevasEvaluaciones.length}.\n\n` + (logErrores.length > 0 ? `Alertas:\n${logErrores.slice(0,5).join('\n')}` : ""));
  } else { ui.alert(`No se generó nada.\n` + logErrores.join('\n')); }
}

function calcularDuracion(iObj, fObj, tz, esCorta) {
  try {
    let sI = "??", sF = "??";
    if (iObj instanceof Date) { sI = Utilities.formatDate(iObj, tz, "HH:mm"); sF = Utilities.formatDate(fObj, tz, "HH:mm"); } 
    else { sI = iObj.substring(0,5); sF = fObj.substring(0,5); }
    let m = esCorta ? 45 : 90;
    return `⏱️ ${sI}-${sF} (${m}m)` + (esCorta ? " ❗" : "");
  } catch (e) { return "⏱️ Hora no definida"; }
}

// NUEVA FIRMA CON 2 VARIABLES EXTRA
function verificarTopesEnMemoria(curso, fObj, tipo, lim, hist, nuevas, asignaturaActual, exentas) {
  // Si la materia actual es exenta, se aprueba inmediatamente el espacio
  if (exentas && exentas.includes(asignaturaActual)) return false;

  const mD = lim[curso] ? lim[curso].maxDia : 2; 
  const mO = lim[curso] ? (lim[curso].maxOtras !== undefined ? lim[curso].maxOtras : mD) : 2; 
  const mE = lim[curso] ? lim[curso].maxEscritas : 2; const mT = lim[curso] ? lim[curso].maxTotal : 4;
  const gw = (d) => { const x = new Date(d); x.setHours(0,0,0,0); x.setDate(x.getDate() + 4 - (x.getDay()||7)); return Math.ceil((((x - new Date(x.getFullYear(),0,1))/8.64e7)+1)/7); };
  const sO = gw(fObj); const aO = fObj.getFullYear(); const fS = fObj.toISOString().split('T')[0];
  let cDE = 0, cDO = 0, cST = 0, cSE = 0;

  hist.concat(nuevas).forEach(f => {
    if (f[2] && f[2].toString().trim() === curso && f[1]) {
      
      // IGNORAR MATERIAS EXENTAS EN EL CONTEO HISTÓRICO
      if (exentas && exentas.includes(f[3].toString().trim())) return;

      const fe = new Date(f[1]); const feS = fe.toISOString().split('T')[0];
      if (feS === fS) { if (f[4] === 'Escrita') cDE++; else cDO++; }
      if (gw(fe) === sO && fe.getFullYear() === aO) { cST++; if (f[4] === 'Escrita') cSE++; }
    }
  });
  return (tipo === 'Escrita' && cDE >= mD) || (tipo !== 'Escrita' && cDO >= mO) || (cST >= mT) || (tipo === 'Escrita' && cSE >= mE); 
}

function instalarTriggerRecordatorios() {
  const ui = SpreadsheetApp.getUi();
  const emailActual = verificarPermisos().email; // Verifica permisos básicos
  
  // Limpiamos triggers anteriores para evitar envíos duplicados
  const triggers = ScriptApp.getProjectTriggers();
  triggers.forEach(t => {
    if (t.getHandlerFunction() === 'enviarRecordatoriosEvaluaciones') {
      ScriptApp.deleteTrigger(t);
    }
  });
  
  // Instalamos el reloj interno: Corre todos los días entre 08:00 y 09:00 AM
  ScriptApp.newTrigger('enviarRecordatoriosEvaluaciones')
    .timeBased()
    .everyDays(1)
    .atHour(8)
    .create();
    
  ui.alert("✅ Sistema de Recordatorios Activado\n\nEl servidor revisará la agenda todos los días a las 08:00 AM y enviará los correos a los profesores correspondientes.");
}

function enviarRecordatoriosEvaluaciones() {
  const hoy = new Date();
  const diaSemana = hoy.getDay();
  
  // Bloqueo extra por seguridad: No ejecutar sábados (6) ni domingos (0)
  if (diaSemana === 0 || diaSemana === 6) return;

  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const sheetEval = ss.getSheetByName('Evaluaciones');
  if (!sheetEval) return;
  
  const datos = sheetEval.getDataRange().getValues().slice(1);
  const tz = ss.getSpreadsheetTimeZone();
  
  // Calculamos la fecha que será en exactamente 14 días más
  const fechaObjetivo = new Date(hoy.getTime());
  fechaObjetivo.setDate(hoy.getDate() + 14);
  const fechaObjetivoStr = Utilities.formatDate(fechaObjetivo, tz, "yyyy-MM-dd");

  datos.forEach(fila => {
    // Fila[1]=Fecha, Fila[5]=Profesor (Email)
    if (!fila[1] || !fila[5]) return; 
    
    const fechaEval = new Date(fila[1]);
    const fechaEvalStr = Utilities.formatDate(fechaEval, tz, "yyyy-MM-dd");
    
    // Si la evaluación es en exactamente 14 días...
    if (fechaEvalStr === fechaObjetivoStr) {
      const curso = fila[2];
      const asignatura = fila[3];
      const tipo = fila[4];
      const emailProfesor = fila[5];
      
      // Calculamos la fecha de entrega límite (7 días antes de la prueba)
      const fechaLimite = new Date(fechaEval.getTime());
      fechaLimite.setDate(fechaEval.getDate() - 7);
      
      const fechaLimiteStr = Utilities.formatDate(fechaLimite, tz, "dd/MM/yyyy");
      const fechaAmigable = Utilities.formatDate(fechaEval, tz, "dd/MM/yyyy");

      const asunto = `⏳ Recordatorio: Entrega de instrumento de evaluación - ${curso} ${asignatura}`;
      const mensaje = `Hola,\n\nEste es un recordatorio automático del sistema de Agenda Escolar.\n\n` +
                      `Tienes una evaluación de tipo "${tipo}" agendada en ${curso} para la asignatura de ${asignatura} el día ${fechaAmigable}.\n\n` +
                      `Te recordamos que debes entregar el instrumento de evaluación a tu coordinador con al menos 7 días de anticipación, es decir, a más tardar el ${fechaLimiteStr}.\n\n` +
                      `Saludos cordiales,\nSistema de Agenda Escolar.`;
      
      try {
        MailApp.sendEmail({
          to: emailProfesor,
          subject: asunto,
          body: mensaje
        });
      } catch(e) {
        console.error(`No se pudo enviar correo a ${emailProfesor}: ${e.message}`);
      }
    }
  });
}

  }));
}

    function obtenerHorasPorCiclo(cursoStr, asigInfo) {
        if (!asigInfo) return 0;
        const c = cursoStr.toLowerCase().normalize("NFD").replace(/[\u0300-\u036f]/g, ""); 
        
        if (c.includes("1") && c.includes("medio") || c.includes("i ") && c.includes("medio") || c.includes("primero medio") || 
            c.includes("2") && c.includes("medio") || c.includes("ii ") && c.includes("medio") || c.includes("segundo medio")) {
            return parseInt(asigInfo.hrsM1) || 0;
        }
        if (c.includes("3") && c.includes("medio") || c.includes("iii") && c.includes("medio") || c.includes("tercero medio") || 
            c.includes("4") && c.includes("medio") || c.includes("iv") && c.includes("medio") || c.includes("cuarto medio")) {
            return parseInt(asigInfo.hrsM2) || 0;
        }
        if ((c.includes("5") || c.includes("6") || c.includes("7") || c.includes("8") || 
             c.includes("quinto") || c.includes("sexto") || c.includes("septimo") || c.includes("octavo")) && c.includes("basico")) {
            return parseInt(asigInfo.hrs2) || 0;
        }
        if ((c.includes("1") || c.includes("2") || c.includes("3") || c.includes("4") || 
             c.includes("primero") || c.includes("segundo") || c.includes("tercero") || c.includes("cuarto")) && c.includes("basico")) {
            return parseInt(asigInfo.hrs1) || 0;
        }
        return parseInt(asigInfo.hrs1) || 0;
    }

    function inicializarTagsReporte() {
        const cursos = [...new Set(datosReporteRaw.map(d => d.curso).filter(Boolean))].sort();
        const asigs = [...new Set(datosReporteRaw.map(d => d.asignatura).filter(Boolean))].sort();
        const profs = [...new Set(datosReporteRaw.map(d => d.profesor).filter(Boolean))].sort();
        const tipos = [...new Set(datosReporteRaw.map(d => d.tipo).filter(Boolean))].sort();

        filtrosReporte.curso = new Set(cursos); filtrosReporte.asignatura = new Set(asigs); filtrosReporte.profesor = new Set(profs); filtrosReporte.tipo = new Set(tipos);

        renderTags('tagsCursos', 'curso', cursos); renderTags('tagsAsignaturas', 'asignatura', asigs); renderTags('tagsProfesores', 'profesor', profs); renderTags('tagsTipos', 'tipo', tipos);
    }

    function renderTags(containerId, categoria, items) {
        const container = document.getElementById(containerId);
        container.innerHTML = items.map(item => `<span class="tag-filtro tag-${categoria} active" onclick="toggleTagReporte(this, '${categoria}', '${item}')">${item}</span>`).join('');
    }

    function toggleTagReporte(element, categoria, valor) {
        if (filtrosReporte[categoria].has(valor)) { filtrosReporte[categoria].delete(valor); element.classList.remove('active');
        } else { filtrosReporte[categoria].add(valor); element.classList.add('active'); }
        procesarGrafico();
    }

    function toggleTodosTags(categoria, marcar) {
        const items = [...new Set(datosReporteRaw.map(d => d[categoria]).filter(Boolean))].sort();
        if (marcar) { items.forEach(item => filtrosReporte[categoria].add(item)); } else { filtrosReporte[categoria].clear(); }
        document.querySelectorAll(`.tag-${categoria}`).forEach(tag => { if (marcar) tag.classList.add('active'); else tag.classList.remove('active'); });
        procesarGrafico();
    }

    function procesarGrafico() {

          const cOk = !d.curso || filtrosReporte.curso.has(d.curso);
          const aOk = !d.asignatura || filtrosReporte.asignatura.has(d.asignatura);
          const pOk = !d.profesor || filtrosReporte.profesor.has(d.profesor);
          const tOk = !d.tipo || filtrosReporte.tipo.has(d.tipo);
          return cOk && aOk && pOk && tOk;
      });
      
      if (chartInstancia) chartInstancia.destroy();

      const contSoloAlertas = document.getElementById('contSoloAlertas');
      
      if (esAuditoria) { 
          canvasContainer.style.display = 'none'; 
          contSoloAlertas.style.display = 'block';
      } else { 
          canvasContainer.style.display = 'block'; 
          contSoloAlertas.style.display = 'none';

      }

          const soloAlertas = document.getElementById('chkSoloAlertas').checked;

                  } else if (horas >= 5) { min = art42.h3_min; max = art42.h3_max; }

                  if (horas === 0 && agendadas === 0) return; 

                  const cumple = (agendadas >= min && agendadas <= max);
                  
                  if (soloAlertas && (horas === 0 || cumple)) return; 

                  let badge = ""; let rowColor = "";

        datosFiltrados.forEach(dato => { const valX = dato[ejeX]; const valL = dato[ejeLeyenda]; if (valX && valL) matriz[valX][valL]++; });

        const datasetsArray = etiquetasLeyenda.map((leyenda, idx) => { return { label: leyenda, data: etiquetasX.map(x => matriz[x][leyenda]), backgroundColor: paletaColores[idx % paletaColores.length], borderRadius: 2 }; });
        
        const ctx = document.getElementById('miGraficoReporte').getContext('2d');
        chartInstancia = new Chart(ctx, { type: 'bar', data: { labels: etiquetasX, datasets: datasetsArray }, options: { responsive: true, maintainAspectRatio: false, plugins: { legend: { display: true, position: 'top' } }, scales: { x: { stacked: true }, y: { stacked: true, beginAtZero: true, ticks: { stepSize: 1 } } } } });
        
        let htmlTabla = `<table class="table table-hover table-striped table-bordered table-sm mb-0 text-nowrap"><thead class="table-light"><tr><th class="text-uppercase text-secondary small">${ejeX}</th>`;
        etiquetasLeyenda.forEach(l => htmlTabla += `<th class="text-uppercase text-secondary small text-center">${l}</th>`); htmlTabla += `<th class="text-uppercase text-secondary small text-center bg-light">Total</th></tr></thead><tbody>`;
        
        etiquetasX.forEach(x => { htmlTabla += `<tr><td class="fw-medium">${x}</td>`; let sumaFila = 0; etiquetasLeyenda.forEach(l => { const val = matriz[x][l]; sumaFila += val; htmlTabla += `<td class="text-center">${val > 0 ? val : '<span class="text-muted opacity-50">-</span>'}</td>`; }); htmlTabla += `<td class="text-center fw-bold bg-light text-primary">${sumaFila}</td></tr>`; });
        htmlTabla += `</tbody></table>`; 
        document.getElementById('contenedorTablaReporte').innerHTML = htmlTabla;
      }
