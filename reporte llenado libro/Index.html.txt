        </div>

            <input class="form-check-input" type="checkbox" id="filtroPendientes" onchange="actualizarVista()">

      </div>

  <script src="https://cdn.jsdelivr.net/npm/bootstrap@5.3.0/dist/js/bootstrap.bundle.min.js"></script>
  <script>
    let datosGlobales = [];

    document.addEventListener('DOMContentLoaded', cargarDatos);

    function cargarDatos() {
      google.script.run.withSuccessHandler(res => {
        datosGlobales = res;
        actualizarVista();
      }).obtenerDatosBD();
    }

    // Transformar fecha (DD/MM/YYYY) a formato legible y extraer el mes para agrupar
    function procesarFecha(fechaStr) {
      const partes = fechaStr.split('/');
      if (partes.length !== 3) return { mesLabel: "Desconocido", mesId: 0, diaTexto: fechaStr };
      
      const diaNum = parseInt(partes[0], 10);
      const mesNum = parseInt(partes[1], 10) - 1;
      const anioNum = parseInt(partes[2], 10);
      const d = new Date(anioNum, mesNum, diaNum);
      
      const dias = ["Domingo", "Lunes", "Martes", "Miércoles", "Jueves", "Viernes", "Sábado"];
      const meses = ["Enero", "Febrero", "Marzo", "Abril", "Mayo", "Junio", "Julio", "Agosto", "Septiembre", "Octubre", "Noviembre", "Diciembre"];
      
      return {
        mesLabel: `${meses[mesNum]} ${anioNum}`,
        mesId: anioNum * 100 + mesNum, // ID numérico para ordenar cronológicamente (ej. 202602)
        diaTexto: `${dias[d.getDay()]} ${diaNum}`
      };
    }

    function actualizarVista() {
      const contenedor = document.getElementById('contenedorPrincipal');
      const filtroProblemas = document.getElementById('filtroPendientes').checked;
      const busqueda = document.getElementById('buscador').value.toLowerCase();
      
      let contadorCriticos = 0;
      let contadorPendientes = 0;
      const mesesMap = new Map();

      datosGlobales.forEach(item => {
        const req = parseInt(item.Requeridas) || 0;
        if (req === 0) return; 

        const esCritico = item.Critico;
        const esPendiente = parseInt(item.Firma) < req;
        
        if (esCritico) contadorCriticos++;
        else if (esPendiente) contadorPendientes++;

        if (filtroProblemas && !esCritico && !esPendiente) return;
        if (busqueda && !item.Profesor.toLowerCase().includes(busqueda)) return;

        const infoFecha = procesarFecha(item.Fecha);
        item.diaTexto = infoFecha.diaTexto; // Guardamos "Lunes 23" en el item
        
        if (!mesesMap.has(infoFecha.mesId)) {
          mesesMap.set(infoFecha.mesId, { label: infoFecha.mesLabel, profesores: {} });
        }
        
        const dataMes = mesesMap.get(infoFecha.mesId);
        if (!dataMes.profesores[item.Profesor]) dataMes.profesores[item.Profesor] = [];
        dataMes.profesores[item.Profesor].push(item);
      });

      document.getElementById('countCriticos').innerText = contadorCriticos;
      document.getElementById('countPendientes').innerText = contadorPendientes;
      document.getElementById('totalRegistros').innerText = datosGlobales.length;

      if (mesesMap.size === 0) {
        contenedor.innerHTML = `<div class="alert alert-success text-center fw-bold">No se encontraron registros.</div>`;
        return;
      }

      // Ordenar meses del más reciente al más antiguo
      const sortedMesIds = Array.from(mesesMap.keys()).sort((a, b) => b - a);
      let htmlAcordeon = `<div class="accordion" id="acordeonMeses">`;

      sortedMesIds.forEach((mesId, index) => {
        const mesInfo = mesesMap.get(mesId);
        const isOpen = index === 0; 
        
        let profHtml = "";
        for (const prof in mesInfo.profesores) {
          
          // Ordenar los días dentro del profesor (opcional, asume que vienen en orden desde Sheets)
          let filas = mesInfo.profesores[prof].map(reg => {
            const crit = reg.Critico;
            const pend = parseInt(reg.Firma) < parseInt(reg.Requeridas);
            const statusClass = crit ? 'critico' : (pend ? 'pendiente' : '');

            return `
              <div class="status-row ${statusClass}">
                <div style="width: 130px;"><span class="badge badge-date">${reg.diaTexto}</span></div>
                <div class="icon-box flex-grow-1 px-3">
                  <div class="d-flex align-items-center gap-1 ${crit ? 'text-error fw-bold' : ''}">
                    <span class="material-symbols-outlined state-icon">${crit ? 'report' : 'menu_book'}</span>
                    <small>${reg.Creadas}/${reg.Requeridas}</small>
                  </div>
                  <div class="d-flex align-items-center ${reg.Contenido > 0 ? 'text-ok opacity-50' : 'text-error'}">
                    <span class="material-symbols-outlined state-icon">description</span>
                  </div>
                  <div class="d-flex align-items-center ${reg.Asistencia > 0 ? 'text-ok opacity-50' : 'text-error'}">
                    <span class="material-symbols-outlined state-icon">group</span>
                  </div>
                </div>
                <div class="small fw-bold ${crit || pend ? 'text-danger' : 'text-success'}">
                  ${crit ? 'SIN CLASES' : (pend ? 'PENDIENTE' : 'AL DÍA')}
                </div>
              </div>`;
          }).join('');

          profHtml += `<div class="profesor-card shadow-sm mb-4"><div class="px-4 py-2 bg-secondary text-white"><small class="fw-bold">${prof}</small></div>${filas}</div>`;
        }

        htmlAcordeon += `
          <div class="accordion-item border-0 mb-3 shadow-sm" style="border-radius: 10px; overflow: hidden;">
            <h2 class="accordion-header" id="heading-${mesId}">
              <button class="accordion-button ${isOpen ? '' : 'collapsed'}" type="button" data-bs-toggle="collapse" data-bs-target="#collapse-${mesId}">
                📅 ${mesInfo.label}
              </button>
            </h2>
            <div id="collapse-${mesId}" class="accordion-collapse collapse ${isOpen ? 'show' : ''}" data-bs-parent="#acordeonMeses">
              <div class="accordion-body">${profHtml}</div>
            </div>
          </div>`;
      });

      htmlAcordeon += `</div>`;
      contenedor.innerHTML = htmlAcordeon;
    }
  </script>
